\begin{frame}{The Math Looks Incredible}
  \textbf{Inline math:} The quadratic formula is $x = \frac{-b \pm \sqrt{b^2 - 4ac}}{2a}$.
  \vspace{0.3cm}

  \textbf{Display math} (Stokes' Theorem):
  \begin{equation}
    \oint_C \mathbf{F} \cdot d\mathbf{r} = \iint_S (\nabla \times \mathbf{F}) \cdot d\mathbf{S}
  \end{equation}
  \vspace{0.1cm}
  \begin{quip}
    Type this as code and it compiles to absolute perfection ---
    no clicking tiny equation editor buttons, no hunting for the $\Sigma$ symbol,
    no fighting with fraction formatting.
  \end{quip}
\end{frame}

\begin{frame}{How to Debug Like a Pro}
  \begin{columns}
    \column{0.55\textwidth}
      \begin{itemize}
        \item When Overleaf flashes red, \textbf{don't panic.}
        \item Ignore the jargon. Find \texttt{l.42} --- that's your broken line!
        \item \textbf{Undefined control sequence} $\rightarrow$ Typo.
        \item \textbf{Missing \$ inserted} $\rightarrow$ Math mode.
        \item \textbf{Runaway argument} $\rightarrow$ Missing \texttt{\}}.
      \end{itemize}
    \column{0.42\textwidth}
      \begin{errorlog}
        ! Undefined control sequence.\\
        l.42 \textbackslash textbfs\{hello\}\\[0.2cm]
        \color{gray!60}! Missing \$ inserted.\\
        l.67 E = mc\^{}2\\[0.2cm]
        \color{codered!60}$\rightarrow$ Fix: line 42 typo.
      \end{errorlog}
      \vspace{0.2cm}
      \begin{protip}
        Copy the red text. Paste into ChatGPT.
        Ask for the fix. Done.
      \end{protip}
